57# v4 (16 sep 2026): el tablero suma dos 'X'. 

58# 

59# Comer una X da +50 y sube el multiplicador un escalón (x2, x3, 

60# ...), de forma PERMANENTE y sin resetearse. El multiplicador 

61# escala solo los puntos de comida (digito x 100 x multiplicador); 

62# no escala los +50 de la X, el +1 por movimiento ni las 

63# penalizaciones de -500. La X no hace crecer la víbora y cada 

64# jugador tiene su propio multiplicador. 

65# 

66# Cuánto vale subir un escalón: 

67# cada comida futura rinde `digito x 100` puntos EXTRA por cada 

68# escalón. Es decir, el valor de la X depende de cuántas comidas 

69# nos queden por comer (más turnos restantes = más valiosa) y 

70# NO del multiplicador actual en términos absolutos... pero sí 

71# en términos RELATIVOS: con un multiplicador alto, cada comida 

72# ya vale mucho, así que conviene priorizar comida antes que ir 

73# a buscar otra X. Por eso dividimos por el multiplicador.

89def bonus_value(remaining_moves, multiplier=1): 

90 """ 

91 Cuánto vale (en la escala de score interna, donde una comida 

92 alcanzable suma 1000) ir a comer una X ahora mismo. 

93 

94 Crece con los turnos que quedan —una X al principio de la 

95 partida escala muchas comidas; una X a 5 turnos del final casi 

96 no escala nada— y baja a medida que sube el multiplicador, 

97 porque con x4 ya encima conviene gastar los turnos comiendo 

98 dígitos en vez de juntando más X. 

99 """ 

100 

101 if remaining_moves is None: 

102 remaining_moves = 100 

103 

104 expected_foods = max(0, remaining_moves) / BONUS_STEPS_PER_FOOD 

105 

106 value = (250 + 250 * expected_foods) / max(1, multiplier) 

107 

108 return max(BONUS_MIN_SCORE, min(BONUS_MAX_SCORE, value))

152def find_snakes(rows, side): 

153 """ 

154 Encuentra cabeza propia, cabeza rival, comida, obstáculos, 

155 dígitos y bonus. 

156 dígitos, bonus y comida de crash (v7). 

157 

158 La comida "clásica" (*) se devuelve en `food`. Los dígitos 

159 ('1'-'9', comida de la v3) se devuelven aparte, en `digits`: 

160 un diccionario {dígito: [posiciones]}, porque no todo dígito 

161 en el tablero sirve — solo el que corresponda comer ahora 

162 según el orden ascendente cíclico. 

163 

164 Las 'X' (v4) van en `bonuses`: son celdas SEGURAS (se pisan 

165 sin chocar) que dan +50 y suben el multiplicador un escalón, 

166 de forma permanente. Ojo: nunca deben terminar en 

167 `obstacles`, o el bot las esquivaría. 

168 

169 v7: comida de crash (Ⓐ / Ⓑ, Unicode U+24B6 / U+24B7). 

170 - La del RIVAL es comida que podemos comer: +100 y crecemos. 

171 Va en `crash_food_enemy`. 

172 - La PROPIA la podemos pisar para limpiarla (denegarla al 

173 rival): no da puntos ni crecimiento. No va como obstáculo. 

174 """ 

175 

176 enemy = "B" if side == "A" else "A" 

177 

178 # Comida de crash: Ⓐ (U+24B6) y Ⓑ (U+24B7). 

179 # El servidor podría usar variantes minúsculas ⓐ (U+24D0) / ⓑ (U+24D1). 

180 own_crash_chars = {"\u24b6", "\u24d0"} if side == "A" else {"\u24b7", "\u24d1"} 

181 enemy_crash_chars = {"\u24b7", "\u24d1"} if side == "A" else {"\u24b6", "\u24d0"} 

182 

183 own_head = None 

184 enemy_head = None 

185 food = [] 

186 digits = {} 

187 bonuses = [] 

188 obstacles = set() 

189 crash_food_enemy = [] 

190 own_crash_food = [] 

191 

192 for r, row in enumerate(rows): 

193 for c, cell in enumerate(row): 

194 

195 if cell == side: 

196 own_head = (r, c) 

197 

198 elif cell == enemy: 

199 enemy_head = (r, c) 

200 

201 elif cell == "*": 

202 food.append((r, c)) 

203 

204 elif cell.isdigit() and cell != "0": 

205 digits.setdefault(cell, []).append((r, c)) 

206 

207 elif cell in BONUS_CHARS: 

208 bonuses.append((r, c)) 

209 

210 elif cell in enemy_crash_chars: 

211 # Comida de crash del rival: la podemos comer (+100) 

212 crash_food_enemy.append((r, c)) 

213 

214 elif cell in own_crash_chars: 

215 # Nuestra comida de crash: la podemos pisar para 

216 # limpiarla (denegar al rival), sin obtener puntos. 

217 # NO es obstáculo ni comida — solo pisable. 

218 own_crash_food.append((r, c)) 

219 

220 elif cell in "ab#": 

221 obstacles.add((r, c)) 

222 

223 # Celdas con letras del rival (cabeza + cuerpo) que no 

224 # son crash food: son obstáculo. Pero la cabeza propia 

225 # y su cuerpo solo se detectan arriba por su letra exacta. 

226 

227 # La cabeza propia es el punto de partida, 

228 # así que no debe considerarse un obstáculo. 

229 if own_head in obstacles: 

230 obstacles.remove(own_head) 

231 

232 return ( 

233 own_head, enemy_head, food, obstacles, digits, bonuses, 

234 crash_food_enemy, own_crash_food 

235 )

263def bfs_path(rows, start, goal, blocked): 

264 """ 

265 Busca el camino más corto entre start y goal. 

266 

267 Devuelve una lista de posiciones. 

268 Si no existe camino, devuelve None. 

269 """ 

270 

271 if start == goal: 

272 return [] 

273 

274 queue = deque([start]) 

275 previous = {start: None} 

276 

277 while queue: 

278 

279 current = queue.popleft() 

280 

281 for _, nxt in neighbors(rows, current): 

282 

283 if nxt in previous: 

284 continue 

285 

286 if nxt in blocked and nxt != goal: 

287 continue 

288 

289 previous[nxt] = current 

290 

291 if nxt == goal: 

292 path = [] 

293 node = nxt 

294 

295 while node != start: 

296 path.append(node) 

297 node = previous[node] 

298 

299 path.reverse() 

300 return path 

301 

302 queue.append(nxt) 

303 

304 return None

309 Calcula cuántas casillas puede alcanzar desde una posición. 

310 Sirve para evitar meternos en zonas cerradas. 

311 """ 

312 

313 if start in blocked: 

314 return 0 

315 

316 queue = deque([start]) 

317 visited = {start} 

318 

319 while queue: 

320 

321 current = queue.popleft() 

322 

323 for _, nxt in neighbors(rows, current):

326 continue 

327 

328 if nxt in blocked: 

329 continue 

330 

331 visited.add(nxt) 

332 queue.append(nxt) 

333 

334 return len(visited)

337def trace_own_body(rows, head, body_cells, neck_hint=None, limit=20000): 

338 """ 

339 Reconstruye el orden real del propio cuerpo, de la cabeza a la 

340 cola. 

341 

342 El tablero no dice en qué orden van los segmentos (todos se ven 

343 igual, como `a`), así que lo deducimos: como el cuerpo de una 

344 víbora es un camino simple (sin bifurcaciones ni cruces 

345 consigo mismo), caminamos desde la cabeza pasando por cada 

346 celda del cuerpo una sola vez. Casi siempre hay un único camino 

347 posible; cuando hay más de una opción, probamos primero la 

348 celda que le queda con MENOS salidas libres — así evitamos 

349 quedarnos sin por dónde seguir más adelante y tener que volver 

350 atrás. 

351 

352 Ojo con un caso ambiguo: si la víbora está enroscada de forma 

353 que la cabeza queda pegada a DOS celdas de su propio cuerpo a 

354 la vez (la cola real Y el cuello, el segmento inmediatamente 

355 anterior a la cabeza), hay dos caminos igual de válidos 

356 geométricamente y no hay forma de saber cuál es cuál mirando 

357 solo el tablero de este turno — hace falta memoria del 

358 movimiento anterior. `neck_hint` es justamente eso: si sabemos 

359 con certeza qué celda es el cuello (porque ahí estaba nuestra 

360 propia cabeza el turno pasado), se la pasamos para arrancar el 

361 camino por el lado correcto sin tener que adivinar. El error, 

362 si no se resuelve, es conservador: como mucho subestima cuánto 

363 se libera la cola, nunca al revés. 

364 

365 Si el cuerpo está tan enroscado que no se completa el camino 

366 dentro del presupuesto de pasos, devolvemos el mejor camino 

367 parcial encontrado y agregamos el resto de las celdas al final, 

368 en cualquier orden. No rompe nada: esas celdas, al no tener un 

369 orden confiable, terminan tratándose como si nunca se liberaran 

370 (la misma suposición conservadora que ya usábamos antes de 

371 esta mejora). 

372 """ 

373 

374 if not body_cells: 

375 return [head] 

376 

377 path = [head] 

378 seen = {head} 

379 best = [head] 

380 budget = [limit] 

381 

382 def free_neighbors(cell): 

383 options = [] 

384 for _, nxt in neighbors(rows, cell): 

385 if nxt in body_cells and nxt not in seen: 

386 options.append(nxt) 

387 return options 

388 

389 # Si tenemos el dato del cuello, forzamos el primer paso ahí 

390 # cuando sea una opción válida — resuelve la ambigüedad de raíz 

391 # en vez de dejarla en manos de la heurística. 

392 if neck_hint is not None and neck_hint in body_cells: 

393 path.append(neck_hint) 

394 seen.add(neck_hint) 

395 best = list(path) 

396 

397 def extend(): 

398 

399 if len(path) - 1 == len(body_cells): 

400 return True 

401 

402 budget[0] -= 1 

403 

404 if budget[0] < 0: 

405 return False 

406 

407 options = free_neighbors(path[-1]) 

408 options.sort(key=lambda n: len(free_neighbors(n))) 

409 

410 for n in options: 

411 

412 path.append(n) 

413 seen.add(n) 

414 

415 if len(path) > len(best): 

416 best[:] = path 

417 

418 if extend(): 

419 return True 

420 

421 path.pop() 

422 seen.discard(n) 

423 

424 return False 

425 

426 if extend(): 

427 return path 

428 

429 remaining = body_cells - set(best) 

430 

431 return best + sorted(remaining)

434def own_body_free_times(body_order): 

435 """ 

436 A partir del orden cabeza->cola (el que devuelve trace_own_body), 

437 calcula en cuántos de NUESTROS PROPIOS movimientos se libera 

438 cada celda del cuerpo. 

439 

440 La cola se libera en el próximo movimiento (1), la celda antes 

441 de la cola en 2, y así hasta la celda pegada a la cabeza. Esto 

442 asume que no comemos nada mientras tanto — si comemos, la cola 

443 NO se mueve ese turno, así que es una estimación optimista, no 

444 una garantía exacta. Por eso este cálculo sirve para elegir 

445 ENTRE opciones, no para asumir a ciegas que un camino angosto 

446 siempre va a estar libre a tiempo. 

447 """ 

448 

449 body = body_order[1:] # sin la cabeza 

450 total = len(body) 

451 

452 return { 

453 cell: total - i 

454 for i, cell in enumerate(body) 

455 }

458def reachable_area_dynamic(rows, start, blocked, free_times, max_time=60): 

459 """ 

460 Como reachable_area, pero las celdas del PROPIO cuerpo que 

461 aparecen en `free_times` se consideran alcanzables una vez que 

462 el número de pasos para llegar ahí alcanza el turno en que se 

463 liberan — en vez de bloqueadas para siempre. 

464 

465 Todo lo demás en `blocked` (cuerpo rival, muros, dígitos a 

466 evitar) se sigue tratando como obstáculo permanente: no 

467 tenemos manera confiable de predecir cuándo se libera lo que 

468 no controlamos nosotros. 

469 """ 

470 

471 hard_blocked = blocked - set(free_times) 

472 

473 if start in hard_blocked: 

474 return 0 

475 

476 visited = {start} 

477 frontier = [start] 

478 time = 0 

479 

480 while frontier and time < max_time: 

481 

482 time += 1 

483 next_frontier = [] 

484 

485 for cell in frontier: 

486 for _, nxt in neighbors(rows, cell): 

487 

488 if nxt in visited: 

489 continue 

490 

491 if nxt in hard_blocked: 

492 continue 

493 

494 free_at = free_times.get(nxt) 

495 

496 if free_at is not None and free_at > time: 

497 continue 

498 

499 visited.add(nxt) 

500 next_frontier.append(nxt) 

501 

502 frontier = next_frontier 

503 

504 return len(visited)

524def food_score( 

525 rows, 

526 position, 

527 food, 

528 blocked, 

529 enemy_head, 

530 preferred_target=None, 

531 total_foods=1, 

532 next_food=None 

533): 

534 """ 

535 Puntúa una comida teniendo en cuenta: 

536 - distancia propia 

537 - distancia del rival 

538 - si llega antes 

539 - si es la comida a la que ya veníamos apuntando (para no 

540 dudar entre dos objetivos parecidos turno a turno) 

541 - cuánta comida hay en total en el mapa: si hay poca, 

542 vale la pena viajar lejos; si hay 

543 mucha, conviene ser selectivo e ir a lo rápido/seguro. 

544 """ 

545 

546 own_path = bfs_path( 

547 rows, 

548 position, 

549 food, 

550 blocked 

551 ) 

552 

553 if own_path is None: 

554 return None 

555 

556 own_distance = len(own_path) 

557 

558 # Para calcular la ruta del rival necesita permitir 

559 # que empiece desde su propia cabeza. 

560 enemy_blocked = set(blocked) 

561 

562 if enemy_head is not None: 

563 enemy_blocked.discard(enemy_head) 

564 enemy_path = bfs_path( 

565 rows, 

566 enemy_head, 

567 food, 

568 enemy_blocked 

569 ) 

570 else: 

571 enemy_path = None 

572 

573 if enemy_path is None: 

574 enemy_distance = 999 

575 else: 

576 enemy_distance = len(enemy_path) 

577 

578 score = 1000 

579 

580 # Con poca comida en el mapa  

581 # (1-2), conviene ir por la que haya aunque esté lejos 

582 # no hay de otra. Con comida abundante mejor prioriza 

583 # lo cercano y deja pasar lo lejano, porque 

584 # seguramente aparezca algo mejor más cerca pronto. 

585 if total_foods <= 2: 

586 distance_weight = 15 

587 elif total_foods >= 5: 

588 distance_weight = 40 

589 else: 

590 distance_weight = 25 

591 

592 score -= own_distance * distance_weight 

593 # --- PREVISIÓN DE COMBO --- 

594 if next_food is not None: 

595 # Se simula cuánto costará ir desde el dígito actual hasta el SIGUIENTE 

596 path_to_next = bfs_path(rows, food, next_food, blocked) 

597 if path_to_next is not None: 

598 # Penaliza el trayecto FUTURO. Ponderación (12) para no opacar 

599 # la distancia actual, pero forzará a elegir el mejor ángulo de ataque. 

600 score -= len(path_to_next) * 12 

601 

602 # Quiere llegar antes que el rival. 

603 race_difference = enemy_distance - own_distance 

604 

605 score += race_difference * 35 

606 

607 # Si el rival llega antes, lo penaliza — pero mucho menos 

608 # si esta es la única comida disponible. 

609 if enemy_distance <= own_distance: 

610 score -= 100 if total_foods <= 2 else 300 

611 

612 # Si llega claramente antes, premiamos. 

613 elif enemy_distance >= own_distance + 3: 

614 score += 250 

615 

616 # Evita zigzaguear cambiando de objetivo cada turno 

617 # entre dos comidas de puntaje parecido. 

618 if preferred_target is not None and food == preferred_target: 

619 score += 120 

620 

621 return score

624def choose_target( 

625 rows, 

626 head, 

627 foods, 

628 blocked, 

629 enemy_head, 

630 preferred_target=None, 

631 next_food=None 

632): 

633 """Elige la comida más conveniente.""" 

634 

635 best_food = None 

636 best_score = float("-inf") 

637 

638 total_foods = len(foods) 

639 

640 for food in foods: 

641 

642 score = food_score( 

643 rows, 

644 head, 

645 food, 

646 blocked, 

647 enemy_head, 

648 preferred_target, 

649 total_foods, 

650 next_food 

651 ) 

652 

653 if score is None: 

654 continue 

655 

656 if score > best_score or ( 

657 score == best_score and food == preferred_target 

658 ): 

659 best_score = score 

660 best_food = food 

661 

662 return best_food

665def simulate_survival(rows, start_head, blocked, depth, free_in=None): 

666 """ 

667 Simula varios movimientos propios hacia adelante para detectar 

668 si un camino que HOY parece amplio termina cerrándose (un 

669 "cuello de botella" que recién se nota unos turnos más tarde). 

670 

671 free_in (opcional): {celda: en cuántos movimientos propios se 

672 libera}, de own_body_free_times(). Sin esto, el cuerpo propio se 

673 trata como bloqueado para siempre durante toda la simulación (lo 

674 conservador de siempre). Con esto, la cola y lo que va detrás 

675 se van habilitando a medida que avanza la simulación — igual 

676 que pasaría en la partida real. 

677 """ 

678 

679 current_blocked = set(blocked) 

680 current_head = start_head 

681 steps = 0 

682 

683 for step_number in range(1, depth + 1): 

684 

685 if free_in: 

686 for cell, free_at in free_in.items(): 

687 if free_at <= step_number: 

688 current_blocked.discard(cell) 

689 

690 candidates = legal_moves(rows, current_head, current_blocked) 

691 

692 if not candidates: 

693 break 

694 

695 best_next = None 

696 best_area = -1 

697 

698 for _, nxt in candidates: 

699 

700 trial_blocked = current_blocked | {current_head} 

701 area = reachable_area(rows, nxt, trial_blocked) 

702 

703 if area > best_area: 

704 best_area = area 

705 best_next = nxt 

706 

707 current_blocked.add(current_head) 

708 current_head = best_next 

709 steps += 1 

710 

711 final_area = reachable_area(rows, current_head, current_blocked) 

712 

713 return steps, final_area

716def enemy_congestion(position, enemy_cells, radius=3): 

717 """ 

718 Cuenta cuántas celdas del cuerpo/cabeza rival hay a una 

719 distancia Manhattan <= radius de `position`. 

720 

721 Sirve para detectar cuándo el bot se está metiendo (o 

722 quedando) en una zona apretada junto al rival, ANTES de que 

723 el área/movilidad lo note — que solo reacciona una vez que 

724 el espacio ya se redujo. Quedarse dando vueltas pegado a un 

725 tramo largo de cuerpo rival, aunque el resto del tablero 

726 esté vacío, es justamente el patrón que llevó a un choque 

727 real en una partida (mucha comida seguía sin comerse, lejos 

728 de esa esquina, mientras las dos serpientes se apretujaban 

729 ahí durante muchos turnos). 

730 """ 

731 

732 count = 0 

733 

734 for cell in enemy_cells: 

735 

736 distance = abs(position[0] - cell[0]) + abs(position[1] - cell[1]) 

737 

738 if distance <= radius: 

739 count += 1 

740 

741 return count

744def voronoi_score(rows, own_head, enemy_head, blocked): 

745 """ 

746 Estima el control del tablero usando la idea del diagrama de Voronoi: 

747 hace un BFS simultáneo desde ambas cabezas y clasifica cada celda libre 

748 según quién llega primero. 

749 

750 Devuelve (celdas_propias, celdas_rival): cuántas celdas "gana" cada uno. 

751 La diferencia (celdas_propias - celdas_rival) da una medida de control: 

752 positivo = ventaja propia, negativo = ventaja del rival. 

753 

754 Por qué esto importa más que solo mirar el área propia: 

755 podés tener 100 celdas alcanzables, pero si el rival controla las 80 

756 donde está la comida, esa área no te sirve de mucho. El Voronoi lo detecta 

757 porque pondera QUIÉN llega antes a cada zona, no solo si vos podés llegar. 

758 

759 Si no hay rival (enemy_head es None), todas las celdas alcanzables son 

760 "propias" — la función sigue siendo útil como medida de área pura. 

761 

762 Nota sobre tableros con cola que se libera: usamos blocked tal cual se 

763 recibe (el blocked estático del turno), igual que reachable_area. No 

764 modelamos la cola que se va liberando porque hacerlo para DOS jugadores 

765 a la vez introduce suposiciones sobre los movimientos del rival que no 

766 tenemos manera de verificar. Es conservador pero consistente. 

767 """ 

768 

769 if enemy_head is None: 

770 return reachable_area(rows, own_head, blocked), 0 

771 

772 # BFS simultáneo desde ambas cabezas. 

773 # Cada celda registra (distancia, dueño): "A" si llegamos primero 

774 # nosotros, "B" si llega primero el rival, "tie" si empatan. 

775 claimed = {} 

776 queue = deque() 

777 

778 if own_head not in blocked: 

779 queue.append((own_head, 0, "A")) 

780 claimed[own_head] = (0, "A") 

781 

782 if enemy_head not in blocked: 

783 queue.append((enemy_head, 0, "B")) 

784 # Si las dos cabezas están en la misma celda (no puede pasar 

785 # en una partida real, pero lo manejamos por robustez): 

786 if enemy_head == own_head: 

787 claimed[enemy_head] = (0, "tie") 

788 else: 

789 claimed[enemy_head] = (0, "B") 

790 

791 while queue: 

792 

793 pos, dist, owner = queue.popleft() 

794 

795 for _, nxt in neighbors(rows, pos): 

796 

797 if nxt in blocked: 

798 continue 

799 

800 new_dist = dist + 1 

801 

802 if nxt not in claimed: 

803 claimed[nxt] = (new_dist, owner) 

804 queue.append((nxt, new_dist, owner)) 

805 

806 elif claimed[nxt][0] == new_dist and claimed[nxt][1] != owner: 

807 # Mismo paso: empate — ninguno se lleva la celda. 

808 claimed[nxt] = (new_dist, "tie") 

809 

810 own_cells = sum(1 for _, owner in claimed.values() if owner == "A") 

811 enemy_cells = sum(1 for _, owner in claimed.values() if owner == "B") 

812 

813 return own_cells, enemy_cells

831 own_score=None, 

832 crash_count=0, 

833 ): 

834 """ 

835 Decide el próximo movimiento. 

836 Combina comida + bonus + seguridad + espacio disponible. 

837 

838 danger_cells: casillas que se pueden pisar (no bloquean el

852

854 tablero. La podemos pisar para limpiarla (denegar al rival)

856 

857 own_score / crash_count (v7): usados para evaluar el riesgo 

858 de crashear. Con score alto y sin crashes previos, un crash 

859 es devastador (wipe a 0). Con score bajo o muchos crashes, 

860 es menos dramático. 

861 

862 Devuelve (direccion, target_elegido). target_elegido es la 

863 comida hacia la que apunta esa decisión (o None si no hay 

864 ninguna alcanzable) — se guarda para pasarla como 

865 preferred_target la próxima vez y así no dudar entre dos 

866 objetivos parecidos turno a turno. 

867 """ 

868 

869 if danger_cells is None: 

870 danger_cells = set() 

871 

872 if bonus_cells is None: 

873 bonus_cells = [] 

874 

875 if crash_food_enemy is None: 

876 crash_food_enemy = [] 

877 

878 if own_crash_food is None: 

879 own_crash_food = [] 

880 

881 own_letter = rows[head[0]][head[1]] 

882 own_body_letter = own_letter.lower() 

883 own_body_cells = { 

884 (r, c) 

885 for r, row in enumerate(rows) 

886 for c, cell in enumerate(row) 

887 if cell == own_body_letter 

888 } 

889 own_body_length = len(own_body_cells) + 1 

890 

891 neck_hint = None 

892 if current_direction in DIRECTIONS: 

893 dr, dc = DIRECTIONS[current_direction] 

894 neck_hint = (head[0] - dr, head[1] - dc) 

895 

896 own_body_order = trace_own_body(rows, head, own_body_cells, neck_hint) 

897 own_free_times = own_body_free_times(own_body_order) 

898 

899 tail_cells = set() 

900 if ( 

901 len(own_body_cells) > 1 

902 and neck_hint in own_body_cells 

903 and own_body_order[-1] != neck_hint 

904 and all( 

905 abs(first[0] - second[0]) + abs(first[1] - second[1]) == 1 

906 for first, second in zip(own_body_order, own_body_order[1:]) 

907 ) 

908 ): 

909 tail_cells.add(own_body_order[-1]) 

910 

911 moves = legal_moves( 

912 rows, 

913 head, 

914 blocked, 

915 passable_cells=tail_cells, 

916 ) 

917 

918 if not moves: 

919 

920 # Callejón sin salida real: no hay ningún movimiento 

921 # que no choque contra algo. Hay que mandar 

922 # ALGO porque quedarse sin responder cuesta un timeout 

923 # que es peor que perder jugando. 

924 fallback_direction = None 

925 fallback_area = -1 

926 

927 for direction, (dr, dc) in DIRECTIONS.items(): 

928 

929 new_head = (head[0] + dr, head[1] + dc) 

930 

931 if not inside_board(rows, *new_head): 

932 continue 

933 

934 if current_direction in OPPOSITE and direction == OPPOSITE[current_direction]: 

935 continue

938 

939 if area > fallback_area: 

940 fallback_area = area 

941 fallback_direction = direction 

942 

943 if fallback_direction is not None: 

944 print("⚠️ Sin salida segura, jugando la menos mala:", fallback_direction) 

945 return fallback_direction, None 

946 

947 return None, None 

948 

949 # Evita invertir inmediatamente la dirección. 

950 if current_direction in OPPOSITE: 

951 

952 opposite = OPPOSITE[current_direction] 

953 

954 non_reverse = [ 

955 move 

956 for move in moves 

957 if move[0] != opposite 

958 ] 

959 

960 if non_reverse: 

961 moves = non_reverse 

962 

963 best_direction = None 

964 best_target = None 

965 best_score = float("-inf") 

966 

967 # Celdas del CUERPO rival (sin la cabeza, que ya viene aparte 

968 # en enemy_head) — para medir congestión más abajo. 

969 enemy_cells = [] 

970 

971 if enemy_head is not None: 

972 enemy_letter = rows[enemy_head[0]][enemy_head[1]] 

973 enemy_body_letter = enemy_letter.lower() 

974 for r, row in enumerate(rows): 

975 for c, cell in enumerate(row): 

976 if cell == enemy_body_letter: 

977 enemy_cells.append((r, c)) 

978 enemy_cells.append(enemy_head) 

979 

980 for direction, new_head in moves: 

981 

982 # ------------------------------------------------ 

983 # 1. Simula su nueva posición. 

984 # ------------------------------------------------ 

985 

986 simulated_blocked = set(blocked) 

987 

988 # Nuestra antigua cabeza pasa a formar parte 

989 # del cuerpo después de movernos. 

990 simulated_blocked.add(head) 

991

993 # 2. Calcula cuánto espacio tendrá. 

994 # ------------------------------------------------ 

995 

996 area = reachable_area_dynamic( 

997 rows, 

998 new_head, 

999 simulated_blocked, 

1000 own_free_times 

1001 ) 

1002

1004 score = area * 8 

1005 

1006 # Control del tablero (Voronoi): BFS simultáneo desde nuestra 

1007 # nueva posición y la cabeza rival. Cuantas más celdas libres 

1008 # llegamos antes que el rival, mejor — porque eso predice quién 

1009 # va a tener acceso a la comida futura y a las zonas abiertas. 

1010 # 

1011 # El área propia ya captura "cuánto espacio tengo", pero no "cuánto 

1012 # de ese espacio me quedo yo vs el rival". El Voronoi agrega esa 

1013 # dimensión: podés tener 100 celdas alcanzables, pero si el rival 

1014 # controla las 80 donde está la comida, esa ventaja no existe. 

1015 # 

1016 # Peso moderado (4): suficiente para mover la aguja entre opciones 

1017 # casi iguales, sin tapar señales más importantes como la comida

1020 rows, 

1021 new_head, 

1022 enemy_head, 

1023 simulated_blocked 

1024 )

1026 voronoi_advantage = own_voronoi - enemy_voronoi 

1027 score += voronoi_advantage * 4

1029 # Detección de encierro por largo propio: si el área 

1030 # alcanzable es menor que nuestra propia víbora, no importa 

1031 # que "parezca" espacio suficiente — no vamos a caber ahí 

1032 # sin chocarnos contra nuestro propio cuerpo eventualmente. 

1033 # Cuanto más grande la diferencia, peor (y esto es una 

1034 # señal MÁS fuerte y MÁS temprana que el look-ahead de más 

1035 # abajo, que recién detecta el problema al simularlo). 

1036 if area < own_body_length: 

1037 faltante = own_body_length - area 

1038 score -= 5000 + 200 * faltante

1041 # cuesta -500 puntos reales. Lo evitamos con una 

1042 # penalización aún mayor, para que sea de verdad el 

1043 # último recurso (no algo "casi tan malo como cualquier 

1044 # otra cosa"). 

1045 if new_head in danger_cells: 

1046 score -= 700

1049 mobility = len( 

1050 legal_moves( 

1051 rows, 

1052 new_head, 

1053 simulated_blocked 

1054 ) 

1055 ) 

1056 

1057 score += mobility * 30

1060 # 2.b Mira varios turnos hacia adelante: ¿este 

1061 # camino se termina cerrando solo, aunque 

1062 # ahora mismo parezca amplio? 

1063 # ------------------------------------------------

1068 rows, 

1069 new_head, 

1070 simulated_blocked, 

1071 LOOKAHEAD_DEPTH, 

1072 free_in=own_free_times 

1073 )

1077 # cuánto espacio le queda a futuro. 

1078 if steps_survived >= LOOKAHEAD_DEPTH: 

1079 score += future_area * 0.3 

1080 

1081 else: 

1082 # Quedó sin movimientos DENTRO del horizonte simulado: 

1083 # esto es una señal fuerte de encierro.  

1084 score -= (LOOKAHEAD_DEPTH - steps_survived) * 500 

1085 

1086 # ------------------------------------------------ 

1087 # 3. Evita acercarse demasiado al rival. 

1088 # 

1089 # v7: chocar ya no termina la partida, pero sigue 

1090 # costando caro: el primer crash pone el score a 0; 

1091 # los siguientes restan -500. Además perdemos cuerpo 

1092 # (todo excepto 3 celdas). La penalización refleja 

1093 # cuánto duele según el estado actual. 

1094 # ------------------------------------------------ 

1095 

1096 if enemy_head is not None: 

1097 

1098 enemy_distance = ( 

1099 abs(new_head[0] - enemy_head[0]) 

1100 + abs(new_head[1] - enemy_head[1]) 

1101 ) 

1102 

1103 if enemy_distance == 0: 

1104 score -= 10000

1108 # Primer crash con score positivo: perderíamos

1110 score -= max(3000, own_score * 2) 

1111 else:

1113 # Sigue siendo malo pero no catastrófico. 

1114 score -= 2000 

1115 

1116 elif enemy_distance == 1: 

1117 score -= 500

1119 # el turno siguiente. 

1120 score -= 400 

1121

1123 # 4. Busca la mejor comida desde esta posición. 

1124 # ------------------------------------------------ 

1125 

1126 target = choose_target( 

1127 rows, 

1128 new_head, 

1129 foods, 

1130 simulated_blocked, 

1131 enemy_head, 

1132 preferred_target, 

1133 next_food 

1134 )

1143 target, 

1144 simulated_blocked 

1145 )

1153

1155 # si puede conseguir otra.

1159 # muchísimo mejor que acercarse

1166 #

1168 # permanente, así que conviene ir a buscarlas — pero sin

1219 # 

1220 # Cada celda de crash food del rival vale +100 (×mult)

1254 cf_step_weight = min(crash_food_value / 10.0, 25)

1258 if closest_crash == 0:

1264 # 

1265 # Si hay comida de crash NUESTRA en el tablero, el

1267 # estamos cerca y no hay mejor opción, conviene pisarla 

1268 # para limpiarla. Pero es baja prioridad comparada 

1269 # con comer dígitos reales. 

1270 # ------------------------------------------------ 

1271 

1272 if own_crash_food: 

1273 

1274 closest_own_crash = None 

1275 

1276 for ocf in own_crash_food: 

1277 

1278 ocf_path = bfs_path( 

1279 rows, 

1280 new_head, 

1281 ocf, 

1282 simulated_blocked 

1283 ) 

1284 

1285 if ocf_path is None:

1287 

1288 if closest_own_crash is None or len(ocf_path) < closest_own_crash: 

1289 closest_own_crash = len(ocf_path) 

1290 

1291 if closest_own_crash is not None: 

1292 # Incentivo bajo: solo vale la pena si estamos 

1293 # cerca y no hay dígitos accesibles. 

1294 score -= closest_own_crash * 3 

1295 

1296 # Pisarla ahora: moderadamente bueno (deniega 

1297 # al rival sin coste propio). 

1298 if closest_own_crash == 0: 

1299 score += 150 

1300 

1301 # ------------------------------------------------ 

1302 # 5. Penalización fuerte por quedar encerrados. 

1303 # ------------------------------------------------ 

1304 

1305 if area <= 3: 

1306 score -= 1000 

1307 

1308 elif area <= 8: 

1309 score -= 400

1313 # si se puede. 

1314 # ------------------------------------------------

1317 height = len(rows) 

1318 width = len(rows[0]) 

1319

1321 r, 

1322 c, 

1323 height - 1 - r, 

1324 width - 1 - c 

1325 )

1327 if not eats_now:

1349def choose_direction(turn_data): 

1350 """ 

1351 Función puente: Recibe el JSON del turno (turn_data) desde run.py, 

1352 prepara las variables, calcula el siguiente dígito para armar combos, 

1353 y delega la decisión matemática a calculate_direction. 

1354 

1355 v6: cada dígito puede tener 3-5 copias. El bot elige la mejor. 

1356 Al comer cualquier copia, avanza el dígito y todas las 

1357 copias desaparecen. El combo lookahead (next_food) busca 

1358 la copia más cercana del siguiente dígito. 

1359 

1360 v7: comida de crash (Ⓐ/Ⓑ) se extrae del tablero y se pasa 

1361 a calculate_direction. Se trackea crash_count para decidir 

1362 la penalización correcta. 

1363 """ 

1364 game_id = turn_data.get("game_id") 

1365 board = turn_data.get("board") 

1366 side = turn_data.get("side") 

1367

1369 ( 

1370 head, enemy_head, foods, blocked, digits, bonuses, 

1371 crash_food_enemy, own_crash_food 

1372 ) = find_snakes(rows, side) 

1373 

1374 if head is None: 

1375 return "up" # Fallback seguro por si no nos encontramos en el tablero 

1376 

1377 remaining_moves = turn_data.get("remaining_moves") 

1378 multiplier = turn_data.get("multiplier_1" if side == "A" else "multiplier_2") or 1 

1379 

1380 # v7: score propio y crash count 

1381 own_score = turn_data.get("score_1" if side == "A" else "score_2")

1392 # --- LÓGICA DE COMBO (v6-aware): Calcular cuál es el SIGUIENTE dígito ---

1395 next_food_pos = None 

1396 if expected_digit_str is not None: 

1397 next_digit_int = (int(expected_digit_str) % 9) + 1 

1398 next_digit_str = str(next_digit_int)

1401

1413 next_food_pos = digits[next_digit_str][0]

1417 for d, positions in digits.items(): 

1418 if d != expected_digit_str:

1422

1441 )
